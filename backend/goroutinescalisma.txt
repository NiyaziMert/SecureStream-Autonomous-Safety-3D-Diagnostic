package main

import (
	"fmt"
	"sync"
	"time"
)

/*
wg *sync.WaitGroup: Burada * (pointer/işaretçi) işareti hayati önem taşır.
WaitGroup'u kopyalayarak değil, adresini vererek gönderiyoruz.

Neden? Çünkü tüm worker'ların (işçilerin) aynı merkezi sayacı güncellemesi gerekir.
Eğer kopyasını gönderseydik, her işçi kendi hayali sayacını azaltır ve ana programdaki wg.Wait() bunu asla duymazdı

defer: Go'ya özgü muazzam bir anahtar kelimedir. Bu satır hemen çalışmaz;
fonksiyonun işi bittiğinde (yani for döngüsü kapandığında) en son çalışacağını garanti eder.

wg.Done(): WaitGroup sayacını 1 azaltır. "Benim işim bitti, beni artık beklemenize gerek yok" sinyalidir.
*/

func processTask(id int, dataChan <-chan string, wg *sync.WaitGroup) {
	defer wg.Done()
	for item := range dataChan {
		fmt.Printf("Worker %d: %s işleniyor...\n", id, item)
		time.Sleep(time.Millisecond * 500) // Simüle edilmiş ağır iş
	}
}

/*
dataChan <-chan string
İşte "ileri seviye" dediğimiz kısım burası.
Bu bir Directional Channel (Yönlü Kanal) tanımıdır.

chan string: Bu bir kanal (channel) ve içinden string geçecek demek.

<-chan: Bu ok işareti kanalın solundaysa:
Bu kanalın "Sadece Okunabilir" (Receive-only) olduğu anlamına gelir.

Neden chan string yerine <-chan string yazıyoruz?
Eğer sadece chan string yazsaydın, bu fonksiyon kanalın içine veri de gönderebilirdi (dataChan <- "yeni veri").
Ancak <-chan kullanarak Go derleyicisine şunu diyoruz:
Bu processTask fonksiyonu, bu kanalı sadece dinleyebilir, içine asla veri fırlatamaz.


chan string	: Hem veri gönderilebilir hem veri alınabilir (Çift yönlü).
<-chan string :	Sadece veri alınabilir (Read-only).
chan<- string : 	Sadece veri gönderilebilir (Write-only).
*/

func main() {
	// Veri göndermek için bir kanal oluşturuyoruz
	var wg sync.WaitGroup
	taskChannel := make(chan string)

	workerCount := 3
	for i := 1; i <= workerCount; i++ {
		// Her bir worker için sayacı 1 artır
		wg.Add(1)
		go processTask(i, taskChannel, &wg)
	}

	// Kanala veri gönderiyoruz
	tasks := []string{"Log1", "Log2", "Log3", "Log4", "Log5"}
	for _, t := range tasks {
		taskChannel <- t
		//Arka planda bekleyen 3 işçiden hangisi o an boşta ise, tünelin diğer ucundan o veriyi çekip alır.
		//Bu işleme "Fan-out" denir; yani bir iş kaynağını birden fazla işçiye dağıtmak.
	}

	close(taskChannel)
	//İşçilere "Artık yeni veri gelmeyecek, elinizdekiler bitince siz de çıkabilirsiniz" mesajı gönderir.
	wg.Wait()
	//Bu çok kritik. Eğer bu olmasaydı, main fonksiyonu bittiği anda arka plandaki tüm işçiler (işleri bitmese bile) anında öldürülürdü.
	//Programın tamamlanması için işçilere zaman tanıyoruz.
	fmt.Println("Tüm işler bitti.")
}
